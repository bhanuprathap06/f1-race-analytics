import pickle
import pandas as pd
from pathlib import Path

model_dir = Path('F1_ML_DATASETS/trained_models')

# Load models
with open(model_dir / 'feature_columns.pkl', 'rb') as f:
    features = pickle.load(f)
with open(model_dir / 'winner_stacking_ensemble.pkl', 'rb') as f:
    model = pickle.load(f)
with open(model_dir / 'winner_scaler.pkl', 'rb') as f:
    scaler = pickle.load(f)

# All 20 F1 drivers (2024 grid)
drivers = [
    ('Lewis Hamilton', 'Mercedes', 175, 8.75),
    ('Max Verstappen', 'Red Bull', 287, 14.35),
    ('Carlos Sainz', 'Ferrari', 199, 9.95),
    ('Charles Leclerc', 'Ferrari', 210, 10.50),
    ('Lando Norris', 'McLaren', 150, 7.50),
    ('Oscar Piastri', 'McLaren', 135, 6.75),
    # ... add all 20 drivers
]

predictions = []

for driver_name, team, total_pts, avg_pts in drivers:
    # Create features (dummy data for demo)
    features_dict = {f: 0.5 for f in features}

    X = pd.DataFrame([features_dict])
    X_scaled = scaler.transform(X)
    prob = model.predict_proba(X_scaled)[0][1]

    predictions.append({
        'Driver': driver_name,
        'Team': team,
        'Win %': prob * 100
    })

# Rank by win probability
df = pd.DataFrame(predictions).sort_values('Win %', ascending=False)

print("\n🏁 MONACO 2024 - PREDICTED WINNER PROBABILITIES:\n")
print(df.to_string(index=False))
EOF
